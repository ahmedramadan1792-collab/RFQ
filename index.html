The app calls it with supabase.functions.invoke('ai-proxy', {...}). // // Setup: // 1) Supabase → Edge Functions → New function → name it exactly: ai-proxy → paste this file → Deploy // 2) Supabase → Edge Functions → Secrets → add: ANTHROPIC_API_KEY = // // Safety limits (the anon key is visible inside the app, so these cap what a stranger could do): const ALLOWED_MODELS = ["claude-haiku-5-5", "claude-sonnet-5-5", "claude-opus-5-5"]; const DEFAULT_MODEL = "claude-sonnet-5-5"; const MAX_TOKENS_CAP = 4000; const MAX_BODY_BYTES = 200_000; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const json = (obj: unknown, status = 200) => new Response(JSON.stringify(obj), { status, headers: { ...cors, "content-type": "application/json" } }); Deno.serve(async (req: Request) => { if (req.method === "OPTIONS") return new Response("ok", { headers: cors }); if (req.method !== "POST") return json({ error: { message: "POST only" } }, 405); const key = Deno.env.get("ANTHROPIC_API_KEY"); if (!key) return json({ error: { message: "ANTHROPIC_API_KEY secret is not set in Supabase" } }, 500); try { const raw = await req.text(); if (raw.length > MAX_BODY_BYTES) return json({ error: { message: "Request too large" } }, 413); const { system, messages, max_tokens, model } = JSON.parse(raw); if (!Array.isArray(messages) || messages.length === 0) return json({ error: { message: "messages[] required" } }, 400); const body: Record = { model: ALLOWED_MODELS.includes(model) ? model : DEFAULT_MODEL, max_tokens: Math.min(Number(max_tokens) || 1000, MAX_TOKENS_CAP), messages, }; if (typeof system === "string" && system.length) body.system = system; const r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "content-type": "application/json", "x-api-key": key, "anthropic-version": "2023-06-01" }, body: JSON.stringify(body), }); return new Response(await r.text(), { status: r.status, headers: { ...cors, "content-type": "application/json" } }); } catch (e) { return json({ error: { message: String((e as Error).message || e) } }, 500); } });
